cmake_minimum_required(VERSION 3.18)
project(a100_vision_infer VERSION 0.4.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 80)
find_package(Torch REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(CUDAToolkit REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
find_path(NCCL_INCLUDE_DIR nccl.h HINTS ENV NCCL_ROOT PATH_SUFFIXES include)
find_library(NCCL_LIBRARY nccl HINTS ENV NCCL_ROOT PATH_SUFFIXES lib lib64)
if(NOT NCCL_INCLUDE_DIR OR NOT NCCL_LIBRARY)
  message(FATAL_ERROR "Install libnccl-dev or set NCCL_ROOT")
endif()
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
# TORCH_CXX_FLAGS carries the libstdc++ ABI setting and must also reach nvcc.
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} ${TORCH_CXX_FLAGS}")
add_library(avi_core src/engine.cpp src/speculative.cpp src/gdn_chunk.cpp src/flash_prefill.cpp src/kernels.cu src/optimized.cu src/deep_kernels.cu)
target_include_directories(avi_core PUBLIC include third_party ${TORCH_INCLUDE_DIRS} ${NCCL_INCLUDE_DIR})
target_link_libraries(avi_core PUBLIC ${TORCH_LIBRARIES} CUDA::cudart OpenSSL::Crypto ${NCCL_LIBRARY} MPI::MPI_CXX)
add_executable(avi-infer src/main.cpp)
target_link_libraries(avi-infer PRIVATE avi_core)
add_executable(avi-worker src/worker.cpp)
target_link_libraries(avi-worker PRIVATE avi_core)
add_executable(avi-tp-reduce-test tests/tp_reduce_test.cpp)
target_link_libraries(avi-tp-reduce-test PRIVATE avi_core)
add_executable(avi-kernel-test tests/kernel_test.cpp tests/optimized_test.cpp)
target_link_libraries(avi-kernel-test PRIVATE avi_core)
enable_testing()
add_test(NAME cuda_kernels COMMAND avi-kernel-test)
add_executable(avi-json-test tests/json_test.cpp)
target_include_directories(avi-json-test PRIVATE include)
add_test(NAME json_grammar COMMAND avi-json-test)

add_executable(avi-memory-test tests/memory_test.cpp)
target_include_directories(avi-memory-test PRIVATE include third_party)
add_test(NAME memory_budget COMMAND avi-memory-test ${CMAKE_CURRENT_SOURCE_DIR}/references/config.json)
add_executable(avi-speculative-test tests/speculative_test.cpp)
target_include_directories(avi-speculative-test PRIVATE include)
add_test(NAME speculative_decisions COMMAND avi-speculative-test)
